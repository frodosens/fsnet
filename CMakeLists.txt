cmake_minimum_required(VERSION 3.24)
project(fsnet VERSION 0.3.0 LANGUAGES C CXX)
include(CTest)
include(FetchContent)
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
find_package(Threads REQUIRED)
include(cmake/OpenSSL.cmake)
option(FSNET_SANITIZERS "Enable address/undefined sanitizers" OFF)
option(FSNET_TEST_RUBY "Run modern Ruby integration tests when available" ON)
option(FSNET_WITH_POSTGRES "Build the PostgreSQL storage adapter from pinned source" OFF)
if(FSNET_WITH_POSTGRES)
  include(cmake/PostgreSQL.cmake)
endif()
option(FSNET_WITH_LUA "Build the Lua script adapter and host executable" ON)
option(FSNET_WITH_RUBY "Enable the optional external Ruby adapter tests" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(FSNET_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
# Fetch exact source archives; never search /usr/local for application libraries.
FetchContent_Declare(asio URL https://codeload.github.com/chriskohlhoff/asio/tar.gz/refs/tags/asio-1-30-2
  URL_HASH SHA256=755bd7f85a4b269c67ae0ea254907c078d408cce8e1a352ad2ed664d233780e8)
FetchContent_Declare(sqlitecpp URL https://codeload.github.com/SRombauts/SQLiteCpp/tar.gz/refs/tags/3.3.2
  URL_HASH SHA256=5aa8eda130d0689bd5ed9b2074714c2dbc610f710483c61ba6cf944cebfe03af)
FetchContent_Declare(libjuice URL https://codeload.github.com/paullouisageneau/libjuice/tar.gz/refs/tags/v1.6.2
  URL_HASH SHA256=5078176d55042f3ccf3999c2556d84903f7edf80177ce4a7bf59507541e93938)
set(SQLITECPP_RUN_CPPLINT OFF CACHE BOOL "" FORCE)
set(SQLITECPP_RUN_CPPCHECK OFF CACHE BOOL "" FORCE)
set(SQLITECPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SQLITECPP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(ENABLE_LOCALHOST_ADDRESS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(asio sqlitecpp libjuice)
add_library(fsnet SHARED src/database.cpp src/postgres_database.cpp src/file_storage.cpp src/storage_factory.cpp src/async_database.cpp src/entity_host.cpp src/entity_schema.cpp src/entity_router.cpp src/node.cpp src/cluster.cpp src/ice.cpp src/c_api.cpp)
target_include_directories(fsnet PUBLIC include PRIVATE ${asio_SOURCE_DIR}/asio/include)
target_compile_definitions(fsnet PRIVATE ASIO_STANDALONE)
target_link_libraries(fsnet PRIVATE SQLiteCpp juice Threads::Threads fsnet-ssl)
target_compile_options(fsnet PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
if(FSNET_WITH_POSTGRES)
  target_compile_definitions(fsnet PRIVATE FSNET_WITH_POSTGRES)
  target_link_libraries(fsnet PRIVATE fsnet-pq)
endif()
add_executable(fsnet-storage-migrate examples/storage_migrate.cpp)
target_link_libraries(fsnet-storage-migrate PRIVATE fsnet)
add_executable(fsnet-node examples/node.cpp)
target_link_libraries(fsnet-node PRIVATE fsnet)
add_executable(fsnet-relay examples/relay.cpp)
target_link_libraries(fsnet-relay PRIVATE fsnet)
add_executable(fsnet-benchmark examples/benchmark.cpp)
target_link_libraries(fsnet-benchmark PRIVATE fsnet)
if(FSNET_WITH_LUA)
  FetchContent_Declare(lua URL https://codeload.github.com/lua/lua/tar.gz/refs/tags/v5.4.7
    URL_HASH SHA256=5c39111b3fc4c1c9e56671008955a1730f54a15b95e1f1bd0752b868b929d8e3)
  FetchContent_MakeAvailable(lua)
  file(GLOB lua_sources CONFIGURE_DEPENDS ${lua_SOURCE_DIR}/*.c)
  list(FILTER lua_sources EXCLUDE REGEX "/(lua|luac|onelua|ltests|linit|liolib|loslib|ldblib|lcorolib|loadlib)\\.c$")
  # Upstream supports C++ compilation: exceptions unwind C++ objects instead of
  # longjmp crossing host frames. Lua symbols remain private to this adapter.
  set_source_files_properties(${lua_sources} PROPERTIES LANGUAGE CXX)
  add_library(fsnet-lua-vm STATIC ${lua_sources})
  target_include_directories(fsnet-lua-vm PRIVATE ${lua_SOURCE_DIR})
  add_library(fsnet-lua STATIC src/lua_engine.cpp)
  target_include_directories(fsnet-lua PUBLIC include PRIVATE ${lua_SOURCE_DIR})
  target_link_libraries(fsnet-lua PUBLIC fsnet PRIVATE fsnet-lua-vm)
  target_compile_options(fsnet-lua PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
  add_executable(fsnet-script examples/script_host.cpp)
  target_link_libraries(fsnet-script PRIVATE fsnet-lua)
endif()
if(BUILD_TESTING)
  add_executable(fsnet-schema-tests tests/entity_schema.cpp)
  target_link_libraries(fsnet-schema-tests PRIVATE fsnet)
  add_test(NAME fsnet-schema COMMAND fsnet-schema-tests)
  add_executable(fsnet-file-tests tests/file_storage.cpp)
  target_link_libraries(fsnet-file-tests PRIVATE fsnet Threads::Threads)
  add_test(NAME fsnet-files COMMAND fsnet-file-tests)
  set_tests_properties(fsnet-files PROPERTIES TIMEOUT 20)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  find_program(FSNET_OPENSSL_TOOL openssl)
  if(Python3_Interpreter_FOUND AND FSNET_OPENSSL_TOOL)
    add_executable(fsnet-tls-tests tests/tls.cpp)
    target_link_libraries(fsnet-tls-tests PRIVATE fsnet)
    add_test(NAME fsnet-tls COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tls_fixture.py
      ${FSNET_OPENSSL_TOOL} $<TARGET_FILE:fsnet-tls-tests>)
    set_tests_properties(fsnet-tls PROPERTIES TIMEOUT 30)
  endif()
  add_executable(fsnet-routing-tests tests/entity_router.cpp)
  target_link_libraries(fsnet-routing-tests PRIVATE fsnet Threads::Threads)
  add_test(NAME fsnet-routing COMMAND fsnet-routing-tests)
  set_tests_properties(fsnet-routing PROPERTIES TIMEOUT 20)
  add_executable(fsnet-entity-tests tests/entity_host.cpp)
  target_link_libraries(fsnet-entity-tests PRIVATE fsnet Threads::Threads)
  add_test(NAME fsnet-entities COMMAND fsnet-entity-tests)
  set_tests_properties(fsnet-entities PROPERTIES TIMEOUT 20)
  add_executable(fsnet-storage-tests tests/storage.cpp)
  target_link_libraries(fsnet-storage-tests PRIVATE fsnet Threads::Threads)
  add_test(NAME fsnet-storage COMMAND fsnet-storage-tests)
  set_tests_properties(fsnet-storage PROPERTIES TIMEOUT 15)
  add_executable(fsnet-tests tests/core.cpp)
  target_link_libraries(fsnet-tests PRIVATE fsnet Threads::Threads)
  target_include_directories(fsnet-tests PRIVATE ${asio_SOURCE_DIR}/asio/include)
  target_compile_definitions(fsnet-tests PRIVATE ASIO_STANDALONE)
  add_test(NAME fsnet-core COMMAND fsnet-tests)
  set_tests_properties(fsnet-core PROPERTIES TIMEOUT 45)
  if(NOT FSNET_SANITIZERS AND UNIX)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      add_test(NAME fsnet-file-process COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/file_storage_process.py $<TARGET_FILE:fsnet>)
      set_tests_properties(fsnet-file-process PROPERTIES TIMEOUT 25)
      add_test(NAME fsnet-process COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/process_test.py
        $<TARGET_FILE:fsnet> $<TARGET_FILE:fsnet-node>)
      set_tests_properties(fsnet-process PROPERTIES TIMEOUT 30)
    endif()
  endif()
  if(FSNET_WITH_LUA)
    add_executable(fsnet-lua-tests tests/lua_test.cpp)
    target_link_libraries(fsnet-lua-tests PRIVATE fsnet-lua)
    add_test(NAME fsnet-lua COMMAND fsnet-lua-tests)
    set_tests_properties(fsnet-lua PROPERTIES TIMEOUT 30)
  endif()
  if(FSNET_WITH_POSTGRES AND FSNET_POSTGRES_TEST_SERVER AND Python3_Interpreter_FOUND AND FSNET_OPENSSL_TOOL)
    add_executable(fsnet-postgres-tests tests/postgres.cpp)
    target_link_libraries(fsnet-postgres-tests PRIVATE fsnet Threads::Threads)
    set(pg_lua "-")
    if(FSNET_WITH_LUA AND NOT FSNET_SANITIZERS)
      set(pg_lua $<TARGET_FILE:fsnet-script>)
    endif()
    add_test(NAME fsnet-postgres COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/postgres_fixture.py
      ${FSNET_POSTGRES_ROOT} ${FSNET_OPENSSL_TOOL} $<TARGET_FILE:fsnet-postgres-tests> $<TARGET_FILE:fsnet> ${pg_lua}
      ${CMAKE_CURRENT_SOURCE_DIR}/examples/lua/stateful.lua)
    set_tests_properties(fsnet-postgres PROPERTIES TIMEOUT 90)
  endif()
  if(FSNET_WITH_RUBY AND FSNET_TEST_RUBY AND NOT FSNET_SANITIZERS)
    find_program(FSNET_RUBY_EXECUTABLE NAMES ruby3.4 ruby3.3 ruby3.2 ruby)
  endif()
  if(FSNET_WITH_RUBY AND FSNET_TEST_RUBY AND NOT FSNET_SANITIZERS AND FSNET_RUBY_EXECUTABLE)
    execute_process(COMMAND ${FSNET_RUBY_EXECUTABLE} -e "exit(Gem::Version.new(RUBY_VERSION) >= Gem::Version.new('3.2') ? 0 : 1)"
      RESULT_VARIABLE FSNET_MODERN_RUBY)
    if(FSNET_MODERN_RUBY EQUAL 0)
      if(TARGET fsnet-tls-tests)
        add_test(NAME fsnet-tls-ruby COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tls_fixture.py
          ${FSNET_OPENSSL_TOOL} $<TARGET_FILE:fsnet-tls-tests> ${FSNET_RUBY_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tls_ruby.rb)
        set_tests_properties(fsnet-tls-ruby PROPERTIES TIMEOUT 30 ENVIRONMENT "FSNET_LIBRARY=$<TARGET_FILE:fsnet>")
      endif()
      add_test(NAME fsnet-ruby COMMAND ${FSNET_RUBY_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ruby_test.rb)
      set_tests_properties(fsnet-ruby PROPERTIES TIMEOUT 30 ENVIRONMENT "FSNET_LIBRARY=$<TARGET_FILE:fsnet>")
    endif()
  endif()
  if(FSNET_WITH_LUA AND Python3_Interpreter_FOUND AND NOT FSNET_SANITIZERS)
    set(contract_ruby "-")
    if(FSNET_WITH_RUBY AND FSNET_TEST_RUBY AND FSNET_RUBY_EXECUTABLE AND FSNET_MODERN_RUBY EQUAL 0)
      set(contract_ruby ${FSNET_RUBY_EXECUTABLE})
    endif()
    add_test(NAME fsnet-entity-process COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/entity_process.py
      $<TARGET_FILE:fsnet> $<TARGET_FILE:fsnet-script> ${CMAKE_CURRENT_SOURCE_DIR}/examples/lua/stateful.lua)
    set_tests_properties(fsnet-entity-process PROPERTIES TIMEOUT 20)
    add_test(NAME fsnet-entity-file-process COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/entity_process.py
      $<TARGET_FILE:fsnet> $<TARGET_FILE:fsnet-script> ${CMAKE_CURRENT_SOURCE_DIR}/examples/lua/stateful.lua file)
    set_tests_properties(fsnet-entity-file-process PROPERTIES TIMEOUT 20)
    add_test(NAME fsnet-script-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/script_contract.py
      $<TARGET_FILE:fsnet> $<TARGET_FILE:fsnet-script> ${contract_ruby} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/contract.rb)
    set_tests_properties(fsnet-script-contract PROPERTIES TIMEOUT 30)
  endif()
endif()
