cmake_minimum_required(VERSION 3.24)
project(fsnet VERSION 0.2.0 LANGUAGES C CXX)
include(CTest)
include(FetchContent)
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
find_package(Threads REQUIRED)
option(FSNET_SANITIZERS "Enable address/undefined sanitizers" OFF)
option(FSNET_TEST_RUBY "Run modern Ruby integration tests when available" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(FSNET_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
# Fetch exact source archives; never search /usr/local for application libraries.
FetchContent_Declare(asio URL https://codeload.github.com/chriskohlhoff/asio/tar.gz/refs/tags/asio-1-30-2
  URL_HASH SHA256=755bd7f85a4b269c67ae0ea254907c078d408cce8e1a352ad2ed664d233780e8)
FetchContent_Declare(sqlitecpp URL https://codeload.github.com/SRombauts/SQLiteCpp/tar.gz/refs/tags/3.3.2
  URL_HASH SHA256=5aa8eda130d0689bd5ed9b2074714c2dbc610f710483c61ba6cf944cebfe03af)
FetchContent_Declare(libjuice URL https://codeload.github.com/paullouisageneau/libjuice/tar.gz/refs/tags/v1.6.2
  URL_HASH SHA256=5078176d55042f3ccf3999c2556d84903f7edf80177ce4a7bf59507541e93938)
set(SQLITECPP_RUN_CPPLINT OFF CACHE BOOL "" FORCE)
set(SQLITECPP_RUN_CPPCHECK OFF CACHE BOOL "" FORCE)
set(SQLITECPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SQLITECPP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(ENABLE_LOCALHOST_ADDRESS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(asio sqlitecpp libjuice)
add_library(fsnet SHARED src/database.cpp src/node.cpp src/cluster.cpp src/ice.cpp src/c_api.cpp)
target_include_directories(fsnet PUBLIC include PRIVATE ${asio_SOURCE_DIR}/asio/include)
target_compile_definitions(fsnet PRIVATE ASIO_STANDALONE)
target_link_libraries(fsnet PRIVATE SQLiteCpp juice Threads::Threads)
target_compile_options(fsnet PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
add_executable(fsnet-node examples/node.cpp)
target_link_libraries(fsnet-node PRIVATE fsnet)
add_executable(fsnet-relay examples/relay.cpp)
target_link_libraries(fsnet-relay PRIVATE fsnet)
add_executable(fsnet-benchmark examples/benchmark.cpp)
target_link_libraries(fsnet-benchmark PRIVATE fsnet)
if(BUILD_TESTING)
  add_executable(fsnet-tests tests/core.cpp)
  target_link_libraries(fsnet-tests PRIVATE fsnet Threads::Threads)
  target_include_directories(fsnet-tests PRIVATE ${asio_SOURCE_DIR}/asio/include)
  target_compile_definitions(fsnet-tests PRIVATE ASIO_STANDALONE)
  add_test(NAME fsnet-core COMMAND fsnet-tests)
  set_tests_properties(fsnet-core PROPERTIES TIMEOUT 45)
  if(NOT FSNET_SANITIZERS AND UNIX)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      add_test(NAME fsnet-process COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/process_test.py
        $<TARGET_FILE:fsnet> $<TARGET_FILE:fsnet-node>)
      set_tests_properties(fsnet-process PROPERTIES TIMEOUT 30)
    endif()
  endif()
  if(FSNET_TEST_RUBY AND NOT FSNET_SANITIZERS)
    find_program(FSNET_RUBY_EXECUTABLE NAMES ruby3.4 ruby3.3 ruby3.2 ruby)
  endif()
  if(FSNET_TEST_RUBY AND NOT FSNET_SANITIZERS AND FSNET_RUBY_EXECUTABLE)
    execute_process(COMMAND ${FSNET_RUBY_EXECUTABLE} -e "exit(Gem::Version.new(RUBY_VERSION) >= Gem::Version.new('3.2') ? 0 : 1)"
      RESULT_VARIABLE FSNET_MODERN_RUBY)
    if(FSNET_MODERN_RUBY EQUAL 0)
      add_test(NAME fsnet-ruby COMMAND ${FSNET_RUBY_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ruby_test.rb)
      set_tests_properties(fsnet-ruby PROPERTIES TIMEOUT 30 ENVIRONMENT "FSNET_LIBRARY=$<TARGET_FILE:fsnet>")
    endif()
  endif()
endif()
