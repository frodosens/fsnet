if(FSNET_WITH_LUA)
  FetchContent_Declare(lua URL https://codeload.github.com/lua/lua/tar.gz/refs/tags/v5.4.7
    URL_HASH SHA256=5c39111b3fc4c1c9e56671008955a1730f54a15b95e1f1bd0752b868b929d8e3)
  FetchContent_MakeAvailable(lua)
  file(GLOB lua_sources CONFIGURE_DEPENDS ${lua_SOURCE_DIR}/*.c)
  list(FILTER lua_sources EXCLUDE REGEX "/(lua|luac|onelua|ltests|linit|liolib|loslib|ldblib|lcorolib|loadlib)\\.c$")
  include(cmake/lua_pattern_budget.cmake)
  # Upstream supports C++ compilation: exceptions unwind C++ objects instead of
  # longjmp crossing host frames. Lua symbols remain private to this adapter.
  set_source_files_properties(${lua_sources} PROPERTIES LANGUAGE CXX)
  add_library(fsnet-lua-vm STATIC ${lua_sources})
  target_include_directories(fsnet-lua-vm PRIVATE ${lua_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/src)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/scripts/fsnet/entity.lua")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/scripts/fsnet/entity.lua" FSNET_LUA_OBJECT_SOURCE)
  configure_file(cmake/lua_object_api.hpp.in generated/fsnet_lua_object_api.hpp @ONLY)
  add_library(fsnet-lua STATIC src/lua_engine.cpp)
  target_include_directories(fsnet-lua PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE ${lua_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/generated)
  target_link_libraries(fsnet-lua PUBLIC fsnet PRIVATE fsnet-lua-vm)
  set_target_properties(fsnet-lua PROPERTIES EXPORT_NAME lua)
  set_target_properties(fsnet-lua-vm PROPERTIES EXPORT_NAME lua-vm)
  add_library(fsnet::lua ALIAS fsnet-lua)
  install(TARGETS fsnet-lua fsnet-lua-vm EXPORT fsnetTargets ARCHIVE DESTINATION lib)
  target_compile_options(fsnet-lua PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
  add_executable(fsnet-container-benchmark benchmarks/container_benchmark.cpp)
  target_link_libraries(fsnet-container-benchmark PRIVATE fsnet-lua)
  add_executable(fsnet-attribute-benchmark benchmarks/attribute_benchmark.cpp)
  target_link_libraries(fsnet-attribute-benchmark PRIVATE fsnet-lua)
  add_executable(fsnet-service examples/server/service.cpp)
  target_link_libraries(fsnet-service PRIVATE fsnet-lua)
  set_target_properties(fsnet-service PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  add_executable(fsnet-script examples/server/main.cpp)
  target_link_libraries(fsnet-script PRIVATE fsnet-lua)
  set_target_properties(fsnet-script PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  add_executable(fsnet-client examples/client/main.cpp)
  target_link_libraries(fsnet-client PRIVATE fsnet-lua)
  set_target_properties(fsnet-client PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
